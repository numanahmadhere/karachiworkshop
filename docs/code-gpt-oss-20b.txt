function drawBar(svg, data, color){
const g=svg.querySelector('g');
const max=Math.max(...data.map(d=>d.salary));
data.forEach((d,i)=>{
const bar=document.createElementNS('http://www.w3.org/2000/svg','rect');
bar.setAttribute('x',i*70);
bar.setAttribute('y',180-(d.salary/max)*160);
bar.setAttribute('width',50);
bar.setAttribute('height',(d.salary/max)*160);
bar.setAttribute('fill',color);
g.appendChild(bar);
const text=document.createElementNS('http://www.w3.org/2000/svg','text');
text.setAttribute('x',i*70+25);
text.setAttribute('y',180+15);
text.setAttribute('text-anchor','middle');
text.textContent=d.city||d.skill;
g.appendChild(text);
});
}
drawBar(document.querySelectorAll('svg')[0],cityData,'#4caf50');
drawBar(document.querySelectorAll('svg')[1],skillData,'#2196f3');
</script>
